#include <stdio.h>
#include <stdint.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <unistd.h>

#define IP_BASE_ADDR   0x43C00000U // Replace with your Vivado AXI base address
#define MAP_SIZE       0x1000U

#define REG_TEMP_HUM   0x00U
#define REG_MQ2        0x04U
#define REG_FAN_CTRL   0x08U
#define REG_STATUS     0x0CU

int main(void) {
    int fd = open("/dev/mem", O_RDWR | O_SYNC);
    if (fd < 0) {
        perror("open /dev/mem");
        return 1;
    }

    volatile uint32_t *regs = (volatile uint32_t *)mmap(
        NULL, MAP_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, IP_BASE_ADDR
    );
    if (regs == MAP_FAILED) {
        perror("mmap");
        close(fd);
        return 1;
    }

    // Example update: temp = 42.5 C (425), hum = 50.0% (500), mq2 = 350, auto_fan_en = 1
    uint16_t temp_x10 = 425;
    uint16_t hum_x10  = 500;
    regs[REG_TEMP_HUM / 4] = ((uint32_t)hum_x10 << 16) | temp_x10;
    regs[REG_MQ2 / 4]      = 350U;
    regs[REG_FAN_CTRL / 4] = (1U << 8) | 35U;

    uint32_t status = regs[REG_STATUS / 4];
    printf("Status register: 0x%08X (alert=%u)\n", status, status & 1U);

    munmap((void *)regs, MAP_SIZE);
    close(fd);
    return 0;
}
